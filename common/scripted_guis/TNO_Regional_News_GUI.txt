scripted_gui = {

					set_variable = { Asia_Checkbox_var = 2 }

						has_country_flag = Custom_Oceania_Interest

					clr_country_flag = Custom_Oceania_Interest

				else = {

				frame = America_Checkbox_var

		window_name = "News_Button_GUI"
		parent_window_name = event_header

		effects = {
			popup_news_button_click = {
				if = {
					limit = { NOT = { has_country_flag = Region_News_Window_visible } }

					set_country_flag = gui_disable_arrows
				}
				else = {
					clr_country_flag = Region_News_Window_visible
				}
			}
		}

		triggers = {
			gui_alert_visible = { NOT = { has_country_flag = gui_disable_arrows } }
