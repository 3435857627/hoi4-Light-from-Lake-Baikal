on_actions = {
	on_startup = {
		effect = {
			if = {
				limit = { original_tag = COX }
				set_country_flag = Custom_Europe_Interest
				set_country_flag = Custom_Asia_Interest
				set_variable = { Europe_Checkbox_var = 2 }
				set_variable = { Asia_Checkbox_var = 2 }
			}
			if = {
				limit = { original_tag = XBJ }
				set_country_flag = Custom_Asia_Interest
				set_variable = { Asia_Checkbox_var = 2 }
			}
		}
	}
	on_daily = {
		effect = {
			if = {
				limit = { original_tag = COX }
				if = {
					limit = { NOT = { has_country_flag = Custom_Europe_Interest } }
					set_country_flag = Custom_Europe_Interest
					set_variable = { Europe_Checkbox_var = 2 }
				}
				if = {
					limit = { NOT = { has_country_flag = Custom_Asia_Interest } }
					set_country_flag = Custom_Asia_Interest
					set_variable = { Asia_Checkbox_var = 2 }
				}
			}
			if = {
				limit = { original_tag = XBJ }
				if = {
					limit = { NOT = { has_country_flag = Custom_Asia_Interest } }
					set_country_flag = Custom_Asia_Interest
					set_variable = { Asia_Checkbox_var = 2 }
				}
			}
		}
	}
}
