# 旧军队理念升级：1 -> 12 -> 123
COX_old_army_idea_advance = {
	if = {
		limit = {
			has_idea = Transform_the_old_army1
		}
		swap_ideas = {
			remove_idea = Transform_the_old_army1
			add_idea = Transform_the_old_army12
		}
	}
	else_if = {
		limit = {
			has_idea = Transform_the_old_army12
		}
		swap_ideas = {
			remove_idea = Transform_the_old_army12
			add_idea = Transform_the_old_army123
		}
	}
	else = {
		add_ideas = Transform_the_old_army1
	}
}

# 混乱的边境降级：1 -> 2 -> 3

		}

		}

			has_idea = chaotic_band2
		}
		swap_ideas = {

		}
	}
}
